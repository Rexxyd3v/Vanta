cmake_minimum_required(VERSION 3.20)
project(Vanta VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(RAYLIB REQUIRED IMPORTED_TARGET raylib)
pkg_check_modules(ENET REQUIRED IMPORTED_TARGET libenet)
pkg_check_modules(OPUS REQUIRED IMPORTED_TARGET opus)
file(GLOB_RECURSE VANTA_SOURCES CONFIGURE_DEPENDS src/*.cpp)
list(REMOVE_ITEM VANTA_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
add_library(vanta_core STATIC ${VANTA_SOURCES})
target_include_directories(vanta_core PUBLIC src "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(vanta_core PUBLIC PkgConfig::RAYLIB PkgConfig::ENET PkgConfig::OPUS ws2_32 winmm gdi32 opengl32)
add_executable(Vanta src/main.cpp)
target_link_libraries(Vanta PRIVATE vanta_core)
if(WIN32)
    enable_language(RC)
    target_sources(Vanta PRIVATE packaging/windows.rc)
    set_source_files_properties(packaging/windows.rc PROPERTIES INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/packaging")
    set_target_properties(Vanta PROPERTIES WIN32_EXECUTABLE TRUE)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(review_capture EXCLUDE_FROM_ALL tools/review_capture.cpp)
    target_link_libraries(review_capture PRIVATE vanta_core mfplat mfreadwrite mfuuid ole32)
    add_executable(rule_tests tests/rule_tests.cpp)
    add_test(NAME match_rules COMMAND rule_tests)
    add_executable(game_smoke tests/game_smoke.cpp)
    target_link_libraries(game_smoke PRIVATE vanta_core)
    add_test(NAME game_smoke COMMAND game_smoke)
    set_tests_properties(game_smoke PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" TIMEOUT 90)
endif()
